function getMonthItems() {
  const { month, year } = getPeriod();
  return items.filter(it => { const d = new Date(it.date); return d.getMonth() === month && d.getFullYear() === year; });
}

function getFiltered() {
  return getMonthItems().filter(it => {
    if (activeTab !== 'all' && it.type !== activeTab) return false;
    if (activeFilter === 'paid' && it.status !== 'paid') return false;
    if (activeFilter === 'unpaid' && it.status !== 'unpaid') return false;
    return true;
  });
}

function diffDays(due) {
  const t = new Date(); t.setHours(0,0,0,0);
  const d = new Date(due); d.setHours(0,0,0,0);
  return Math.round((d - t) / 86400000);
}

function renderAll() {
  const all = getMonthItems();
  const income = all.filter(i=>i.type==='income').reduce((s,i)=>s+i.amount,0);
  const expAll = all.filter(i=>i.type==='expense');
  const expense = expAll.reduce((s,i)=>s+i.amount,0);
  const unpaid = all.filter(i=>i.status==='unpaid');
  const unpaidSum = unpaid.filter(i=>i.type==='expense').reduce((s,i)=>s+i.amount,0);
  const bal = income - expense;

  document.getElementById('totalIncome').textContent = '฿' + fmt(income);
  document.getElementById('totalExpense').textContent = '฿' + fmt(expense);
  document.getElementById('balance').textContent = (bal < 0 ? '-' : '') + '฿' + fmt(Math.abs(bal));
  document.getElementById('incomeCount').textContent = all.filter(i=>i.type==='income').length + ' รายการ';
  document.getElementById('expenseCount').textContent = expAll.length + ' รายการ';
  document.getElementById('unpaidCount').textContent = unpaid.length + ' รายการค้างชำระ · ฿' + fmt(unpaidSum);

  const overdue = unpaid.filter(i => i.due && diffDays(i.due) < 0);
  const ab = document.getElementById('alertBar');
  if (overdue.length) { ab.style.display = ''; document.getElementById('alertText').textContent = '⚠️ เกินกำหนดชำระ ' + overdue.length + ' รายการ: ' + overdue.map(i=>i.name).join(', '); }
  else ab.style.display = 'none';

  const filtered = getFiltered();
  const tbody = document.getElementById('tableBody');
  const empty = document.getElementById('emptyState');
  if (!filtered.length) { tbody.innerHTML = ''; empty.style.display = ''; return; }
  empty.style.display = 'none';
  const sorted = [...filtered].sort((a,b) => {
    const so = {unpaid:0, pending:1, paid:2};
    if (so[a.status] !== so[b.status]) return so[a.status] - so[b.status];
    return (a.due||'9') > (b.due||'9') ? 1 : -1;
  });
  tbody.innerHTML = sorted.map(rowHTML).join('');
}

function fmt(n) { return Number(n).toLocaleString('th-TH', {minimumFractionDigits:2, maximumFractionDigits:2}); }

function thDate(s) { const d = new Date(s); return d.getDate() + ' ' + MONTHS_TH[d.getMonth()].slice(0,3) + ' ' + (d.getFullYear()+543); }

function esc(s) { return String(s||'').replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;'); }

function statusLabel(s) { return {paid:'ชำระแล้ว', unpaid:'ค้างชำระ', pending:'รอดำเนินการ'}[s]||s; }

function dueTxt(it) {
  if (!it.due) return '<span style="color:var(--text3);font-size:12px">—</span>';
  if (it.status === 'paid') return '<span class="item-due ok">' + thDate(it.due) + '</span>';
  const diff = diffDays(it.due);
  if (diff < 0) return '<span class="item-due overdue">🔴 เกิน ' + Math.abs(diff) + ' วัน<br><small>' + thDate(it.due) + '</small></span>';
  if (diff === 0) return '<span class="item-due overdue">🔴 ครบกำหนดวันนี้</span>';
  if (diff <= 5) return '<span class="item-due soon">⏰ อีก ' + diff + ' วัน<br><small>' + thDate(it.due) + '</small></span>';
  return '<span class="item-due ok">' + thDate(it.due) + '</span>';
}

function rowHTML(it) {
  const amtTxt = it.amount === 0
    ? '<span style="color:var(--text3);font-size:11px">รอยืนยัน</span>'
    : '<div class="amount-val ' + it.type + '">' + (it.type==='income'?'+':'') + '฿' + fmt(it.amount) + '</div>';
  return '<tr><td><div class="item-name">' + esc(it.name) + '</div>' + (it.note?'<div class="item-note">'+esc(it.note)+'</div>':'') + '</td><td><span class="item-cat">' + esc(it.category) + '</span></td><td>' + dueTxt(it) + '</td><td><span class="status-badge ' + it.status + '" onclick="toggleStatus(\'' + it.id + '\')" title="คลิกเพื่อเปลี่ยนสถานะ"><span class="dot"></span>' + statusLabel(it.status) + '</span></td><td class="amount-cell">' + amtTxt + '</td><td><div class="action-btns"><button class="action-btn" onclick="editItem(\'' + it.id + '\')" title="แก้ไข">✎</button><button class="action-btn del" onclick="deleteItem(\'' + it.id + '\')" title="ลบ">✕</button></div></td></tr>';
}

function switchTab(tab) { activeTab = tab; document.querySelectorAll('.tab').forEach(t => t.classList.toggle('active', t.dataset.tab === tab)); renderAll(); }
function setFilter(f) { activeFilter = f; document.querySelectorAll('.filter-chip').forEach(c => c.classList.toggle('active', c.classList.contains('f-'+f))); renderAll(); }

function openModal(id) {
  editId = id || null;
  document.getElementById('modalTitle').textContent = id ? 'แก้ไขรายการ' : 'เพิ่มรายการใหม่';
  if (id) {
    const it = items.find(i => i.id === id);
    setType(it.type);
    document.getElementById('fName').value = it.name;
    document.getElementById('fAmount').value = it.amount;
    document.getElementById('fCat').value = it.category;
    document.getElementById('fDate').value = it.date;
    document.getElementById('fDue').value = it.due || '';
    document.getElementById('fStatus').value = it.status;
    document.getElementById('fNote').value = it.note || '';
  } else {
    setType('expense');
    ['fName','fAmount','fDue','fNote'].forEach(id => document.getElementById(id).value = '');
    document.getElementById('fDate').value = new Date().toISOString().slice(0,10);
    document.getElementById('fStatus').value = 'unpaid';
  }
  document.getElementById('modalBackdrop').classList.add('open');
}
function closeModal() { document.getElementById('modalBackdrop').classList.remove('open'); }
function handleBackdropClick(e) { if (e.target === e.currentTarget) closeModal(); }
function setType(t) { currentType = t; document.getElementById('typeExpense').classList.toggle('active', t==='expense'); document.getElementById('typeIncome').classList.toggle('active', t==='income'); }

function saveItem() {
  const name = document.getElementById('fName').value.trim();
  const amount = parseFloat(document.getElementById('fAmount').value) || 0;
  const date = document.getElementById('fDate').value;
  if (!name || !date) { showToast('⚠️ กรุณากรอกชื่อรายการและวันที่'); return; }
  const obj = { name, amount, category: document.getElementById('fCat').value, date, due: document.getElementById('fDue').value, status: document.getElementById('fStatus').value, note: document.getElementById('fNote').value.trim(), type: currentType };
  if (editId) { const idx = items.findIndex(i=>i.id===editId); items[idx] = {...items[idx], ...obj}; showToast('✓ แก้ไขเรียบร้อย'); }
  else { items.push({...obj, id: Date.now().toString(36)+Math.random().toString(36).slice(2)}); showToast('✓ เพิ่มรายการเรียบร้อย'); }
  save(); closeModal(); renderAll();
}
function editItem(id) { openModal(id); }
function deleteItem(id) { if (!confirm('ยืนยันการลบรายการนี้?')) return; items = items.filter(i=>i.id!==id); save(); renderAll(); showToast('ลบรายการแล้ว'); }
function toggleStatus(id) { const it = items.find(i=>i.id===id); it.status = {paid:'unpaid',unpaid:'pending',pending:'paid'}[it.status]||'paid'; save(); renderAll(); showToast('สถานะ: ' + statusLabel(it.status)); }
function save() { localStorage.setItem('ledger_v3', JSON.stringify(items)); }
